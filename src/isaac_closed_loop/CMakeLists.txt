cmake_minimum_required(VERSION 3.16)
project(ego4p3_isaac_closed_loop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(P3_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(P3_CANDIDATE "${P3_ROOT}/src/new_controller/ldlt")
set(P3_DEPS "${P3_ROOT}/src/third_party")
set(P3_LIBDIR "${P3_DEPS}/suitesparse/usr/lib/x86_64-linux-gnu")
find_library(P3_CHOLMOD cholmod PATHS "${P3_LIBDIR}" NO_DEFAULT_PATH REQUIRED)
add_library(isaac_navigation_backend STATIC "${P3_CANDIDATE}/src/sparse_trial_backend.cpp")
target_include_directories(isaac_navigation_backend PUBLIC "${P3_CANDIDATE}/include" "${CMAKE_CURRENT_SOURCE_DIR}"
  PRIVATE "${P3_DEPS}/amgcl-1.4.9" "${P3_DEPS}/suitesparse/usr/include/suitesparse")
target_compile_definitions(isaac_navigation_backend PUBLIC EGO4P2_STEP2=1 EGO4P0_CURVE_GUARD=1
  EGO4P0_CORNER_SMALL_STEP=1 EGO4P0_COARSE_PC_MODE=2 AMGCL_NO_BOOST)
target_compile_options(isaac_navigation_backend PUBLIC -fno-fast-math -pthread
  PRIVATE -Wall -Wextra -Wno-unknown-pragmas)
target_link_libraries(isaac_navigation_backend PUBLIC Eigen3::Eigen "${P3_CHOLMOD}" yaml-cpp Threads::Threads)
target_link_options(isaac_navigation_backend INTERFACE "-Wl,--disable-new-dtags")
add_executable(isaac_closed_loop_controller controller.cpp)
target_link_libraries(isaac_closed_loop_controller isaac_navigation_backend)
set_target_properties(isaac_closed_loop_controller PROPERTIES BUILD_RPATH "${P3_LIBDIR}" INSTALL_RPATH "${P3_LIBDIR}")
install(TARGETS isaac_closed_loop_controller RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(supervisor_check supervisor_check.cpp)
  target_link_libraries(supervisor_check isaac_navigation_backend)
  set_target_properties(supervisor_check PROPERTIES BUILD_RPATH "${P3_LIBDIR}")
  add_test(NAME supervisor_check COMMAND supervisor_check "${P3_ROOT}/src/new_controller/config/isaac_closed_loop")
endif()

cmake_minimum_required(VERSION 3.16)
project(ego4p3_r05_model LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
get_filename_component(P3_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CORE "${P3_ROOT}/src/new_controller/ldlt")
set(DEPS "${P3_ROOT}/src/third_party")
set(LIBDIR "${DEPS}/suitesparse/usr/lib/x86_64-linux-gnu")
find_library(CHOLMOD cholmod PATHS "${LIBDIR}" NO_DEFAULT_PATH REQUIRED)
add_executable(r05_controller controller.cpp src/verified_start.cpp "${CORE}/src/sparse_trial_backend.cpp")
target_include_directories(r05_controller PRIVATE "${CORE}/include" include
  "${DEPS}/amgcl-1.4.9" "${DEPS}/suitesparse/usr/include/suitesparse")
target_compile_definitions(r05_controller PRIVATE EGO4P2_STEP2=1 EGO4P0_CURVE_GUARD=1
  EGO4P0_CORNER_SMALL_STEP=1 EGO4P0_COARSE_PC_MODE=2 AMGCL_NO_BOOST)
target_compile_options(r05_controller PRIVATE -fno-fast-math -pthread)
target_link_libraries(r05_controller PRIVATE Eigen3::Eigen yaml-cpp Threads::Threads OpenSSL::Crypto "${CHOLMOD}")
target_link_options(r05_controller PRIVATE "-Wl,--disable-new-dtags")
set_target_properties(r05_controller PROPERTIES BUILD_RPATH "${LIBDIR}")

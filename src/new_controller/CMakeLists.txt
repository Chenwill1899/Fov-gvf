get_filename_component(EGO4P2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(NAV_CANDIDATE "${EGO4P2_ROOT}/src/new_controller/ldlt")
set(NAV_DEPS "${EGO4P2_ROOT}/src/third_party")
set(NAV_LIBDIR "${NAV_DEPS}/suitesparse/usr/lib/x86_64-linux-gnu")
find_library(NAV_CHOLMOD cholmod PATHS "${NAV_LIBDIR}" NO_DEFAULT_PATH REQUIRED)
add_library(navigation_ldlt_backend STATIC "${NAV_CANDIDATE}/src/sparse_trial_backend.cpp")
target_include_directories(navigation_ldlt_backend PUBLIC "${NAV_CANDIDATE}/include"
  PRIVATE "${NAV_DEPS}/amgcl-1.4.9" "${NAV_DEPS}/suitesparse/usr/include/suitesparse")
target_link_libraries(navigation_ldlt_backend PUBLIC Eigen3::Eigen "${NAV_CHOLMOD}" yaml-cpp Threads::Threads)
target_link_options(navigation_ldlt_backend INTERFACE "-Wl,--disable-new-dtags")
target_compile_definitions(navigation_ldlt_backend PUBLIC EGO4P2_STEP2=1 EGO4P0_CURVE_GUARD=1
  EGO4P0_CORNER_SMALL_STEP=1 EGO4P0_COARSE_PC_MODE=2 AMGCL_NO_BOOST)
target_compile_options(navigation_ldlt_backend PRIVATE -fno-fast-math -Wall -Wextra -Wno-unknown-pragmas)
add_executable(navigation_input src/navigation_input.cpp)
add_executable(navigation_replay "${NAV_CANDIDATE}/test/navigation_replay.cpp")
add_executable(intrinsic_2d_benchmark "${NAV_CANDIDATE}/test/intrinsic_2d_benchmark.cpp")
add_executable(local_fine_benchmark "${NAV_CANDIDATE}/test/local_fine_benchmark.cpp")
target_link_libraries(local_fine_benchmark navigation_ldlt_backend)
target_compile_options(local_fine_benchmark PRIVATE -fno-fast-math -pthread)
set_target_properties(local_fine_benchmark PROPERTIES CXX_EXTENSIONS OFF BUILD_RPATH "${NAV_LIBDIR}")
target_link_libraries(intrinsic_2d_benchmark navigation_ldlt_backend)
target_compile_options(intrinsic_2d_benchmark PRIVATE -fno-fast-math -pthread)
set_target_properties(intrinsic_2d_benchmark PROPERTIES CXX_EXTENSIONS OFF INSTALL_RPATH "${NAV_LIBDIR}")
target_link_libraries(navigation_replay navigation_ldlt_backend)
set_target_properties(navigation_replay PROPERTIES INSTALL_RPATH "${NAV_LIBDIR}")
target_link_libraries(navigation_input navigation_ldlt_backend)
target_include_directories(navigation_input AFTER PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include/new_controller")
ament_target_dependencies(navigation_input rclcpp sensor_msgs nav_msgs geometry_msgs rosgraph_msgs)
target_compile_options(navigation_input PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
set_target_properties(navigation_input PROPERTIES INSTALL_RPATH "${NAV_LIBDIR}")
add_executable(depth_reader src/depth_reader.cpp)
target_include_directories(depth_reader PRIVATE include)
ament_target_dependencies(depth_reader rclcpp sensor_msgs nav_msgs geometry_msgs rosgraph_msgs)

# Include preprocessing headers and compile settings in the executed identity.
file(GLOB_RECURSE IDENTITY_FILES CONFIGURE_DEPENDS src/*.cpp include/*.hpp)
file(GLOB_RECURSE NAV_IDENTITY CONFIGURE_DEPENDS "${NAV_CANDIDATE}/include/*.hpp")
list(APPEND IDENTITY_FILES ${NAV_IDENTITY} "${NAV_CANDIDATE}/src/sparse_trial_backend.cpp"
  "${EGO4P2_ROOT}/src/new_controller/config/navigation_step2.json" "${NAV_CANDIDATE}/test/navigation_replay.cpp"
  "${NAV_CANDIDATE}/test/fixed_tracking_cases.hpp"
  "${NAV_CANDIDATE}/test/rolling_cases.hpp")
list(APPEND IDENTITY_FILES "${NAV_CANDIDATE}/test/intrinsic_2d_benchmark.cpp")
list(APPEND IDENTITY_FILES "${NAV_CANDIDATE}/test/local_fine_benchmark.cpp")
list(APPEND IDENTITY_FILES "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${IDENTITY_FILES})
set(IDENTITY_JSON "{")
set(IDENTITY_SEPARATOR "")
foreach(INPUT_FILE IN LISTS IDENTITY_FILES)
  file(SHA256 "${INPUT_FILE}" INPUT_SHA256)
  file(RELATIVE_PATH INPUT_RELATIVE "${EGO4P2_ROOT}" "${INPUT_FILE}")
  string(APPEND IDENTITY_JSON "${IDENTITY_SEPARATOR}\"${INPUT_RELATIVE}\":\"${INPUT_SHA256}\"")
  set(IDENTITY_SEPARATOR ",")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build_identity.json" "${IDENTITY_JSON}}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build_identity.json" DESTINATION share/${PROJECT_NAME})
install(TARGETS depth_reader navigation_input navigation_replay intrinsic_2d_benchmark DESTINATION lib/${PROJECT_NAME})

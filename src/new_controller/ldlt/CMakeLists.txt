cmake_minimum_required(VERSION 3.16)
project(new_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_executable(depth_reader src/depth_reader.cpp)
target_include_directories(depth_reader PRIVATE include)
ament_target_dependencies(depth_reader rclcpp sensor_msgs nav_msgs geometry_msgs rosgraph_msgs)
target_link_libraries(depth_reader yaml-cpp Threads::Threads)
target_compile_options(depth_reader PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
# Include preprocessing headers and compile settings in the executed identity.
file(GLOB_RECURSE IDENTITY_FILES CONFIGURE_DEPENDS src/*.cpp include/*.hpp)
list(APPEND IDENTITY_FILES "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${IDENTITY_FILES})
set(IDENTITY_JSON "{")
set(IDENTITY_SEPARATOR "")
foreach(INPUT_FILE IN LISTS IDENTITY_FILES)
  file(SHA256 "${INPUT_FILE}" INPUT_SHA256)
  file(RELATIVE_PATH INPUT_RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${INPUT_FILE}")
  string(APPEND IDENTITY_JSON "${IDENTITY_SEPARATOR}\"src/new_controller/${INPUT_RELATIVE}\":\"${INPUT_SHA256}\"")
  set(IDENTITY_SEPARATOR ",")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build_identity.json" "${IDENTITY_JSON}}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build_identity.json" DESTINATION share/${PROJECT_NAME})
install(TARGETS depth_reader DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
